import "./{{module}}Page.scss";
import {DxDataGrid} from "../../core/datagrid/DxDataGrid";
import {DxDialog} from "../../core/dialog/DxDialog";
import {DxForm} from "../../core/form/DxForm";
import {Constants} from "../../core/common/Constants";

export const {{module}}Page = () => {
    const container = "wrapper";
    const fields = {{module}}Fields();
    const detailFields = {{module}}DetailFields();

    const API_URL = Constants.API_BASE_URL + '{{kebabModule}}s';

    const EVENTS = {

        EDIT: '{{kebabModule}}:edit',
        DELETE: '{{kebabModule}}:delete',
        SAVE: '{{kebabModule}}:save',
        CANCEL: '{{kebabModule}}:cancel',
        SUCCESS: '{{kebabModule}}:success',
        LOADING: '{{kebabModule}}:loading',
    }

    return (
        <div id={container}>
            <DxDataGrid

                    detailFields={detailFields}

            />

            >
                <DxForm

                        fields={fields}
